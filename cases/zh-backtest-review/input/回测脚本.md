# backtest_v3.py（研究员提交的回测脚本，原样贴出）

```python
import pandas as pd
import numpy as np

START, END = "2019-01-01", "2025-12-31"
TOP_N = 50
COST = 0.0          # 交易成本：暂不考虑

# 行情：日频复权收盘价，index=date, columns=code
px = pd.read_parquet("data/daily_close_adj.parquet")
# 股票池：中证1000 成分股（2025-12-31 导出的最新成分名单）
universe = pd.read_csv("data/csi1000_constituents_latest.csv")["code"].tolist()
px = px[universe].loc[START:END]

# 财务：按报告期存储，index=report_period(季末日期), columns=code
pb = pd.read_parquet("data/pb_by_report_period.parquet")
roe = pd.read_parquet("data/roe_by_report_period.parquet")
# 把季度财务数据对齐到日频
pb_d = pb.reindex(px.index, method="ffill")
roe_d = roe.reindex(px.index, method="ffill")
roe_yoy = roe_d - roe_d.shift(252)
pb_d = pb_d.fillna(method="bfill")      # 补齐缺失值
roe_yoy = roe_yoy.fillna(method="bfill")

# 动量：12-1 月
mom = px.shift(21) / px.shift(252) - 1

month_ends = px.resample("M").last().index
month_ends = [d for d in month_ends if d in px.index]

def score(d):
    s = (0.4 * (-pb_d.loc[d]).rank(pct=True)
         + 0.4 * mom.loc[d].rank(pct=True)
         + 0.2 * roe_yoy.loc[d].rank(pct=True))
    return s.dropna()

nav, weights = [1.0], {}
for i in range(len(month_ends) - 1):
    d0, d1 = month_ends[i], month_ends[i + 1]
    picks = score(d0).nlargest(TOP_N).index
    w = pd.Series(1.0 / TOP_N, index=picks)
    # 以 d0 当日收盘价成交，持有到 d1 收盘
    ret = (px.loc[d1, picks] / px.loc[d0, picks] - 1).mean()
    turnover = (w.subtract(weights.get(i - 1, pd.Series(dtype=float)), fill_value=0).abs().sum())
    nav.append(nav[-1] * (1 + ret - COST * turnover))
    weights[i] = w

nav = pd.Series(nav, index=month_ends[: len(nav)])
ann = nav.iloc[-1] ** (12 / (len(nav) - 1)) - 1
sharpe = nav.pct_change().mean() / nav.pct_change().std() * np.sqrt(12)
mdd = (nav / nav.cummax() - 1).min()
print(f"年化 {ann:.1%}  夏普 {sharpe:.2f}  最大回撤 {mdd:.1%}")
```
